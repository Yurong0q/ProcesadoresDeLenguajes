Terminales = { && > < + - ! real cadena entero id write read return = |=
let if ( ) { } for int float boolean string void function ; , eof }

NoTerminales = { I P E E_AND E1 E1_REL E2 E2_ARIT E3 E4 S S_SIMPLE S_COMPUESTA ASIGFUN A L Q X T INI ACTU C F H B K }

Axioma = I

Producciones = {
    I -> P
    P -> S P 
    P -> F P 
    P -> eof ////Expresiones:
    E -> E1 E_AND
    E_AND   -> && E1 E_AND
    E_AND   -> lambda
    E1      -> E2 E1_REL
    E1_REL  -> > E2 E1_REL
    E1_REL  -> < E2 E1_REL
    E1_REL  -> lambda
    E2      -> E3 E2_ARIT
    E2_ARIT -> + E3 E2_ARIT
    E2_ARIT -> - E3 E2_ARIT 
    E2_ARIT -> lambda
    E3  -> ! E3 
    E3  -> id E4    ////Funcion
    E3  -> ( E )
    E3  -> cadena
    E3  -> entero
    E3  -> real 
    E4  -> ( L ) 
    E4  -> lambda ////Sentencias:
    S -> S_SIMPLE
    S -> S_COMPUESTA
    S_SIMPLE -> id ASIGFUN
    S_SIMPLE -> write E ;
    S_SIMPLE -> read id ;
    S_SIMPLE -> return X ;
    ASIGFUN  -> A E ;
    ASIGFUN  -> ( L ) ;
    A -> =
    A -> |=
    L -> E Q
    L -> lambda 
    Q -> , E Q 
    Q -> lambda
    X -> E
    X -> lambda
    S_COMPUESTA -> let T id ;
    S_COMPUESTA -> if ( E ) S_SIMPLE
    S_COMPUESTA -> for ( INI ; E ; ACTU ) { C }
    T -> int
    T -> float
    T -> boolean
    T -> string
    INI  -> id = E
    INI  -> lambda
    ACTU -> id A E ; 
    ACTU -> lambda
    C -> S C 
    C -> lambda ////Declaracion de funciones:
    F -> function H id ( B ) { C }
    H -> T
    H -> void
    B -> T id K 
    B -> void
    K -> , T id K 
    K -> lambda
}
